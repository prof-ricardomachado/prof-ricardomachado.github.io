\begin{align*} \int_{\gamma_1} \vec{F} \cdot d\vec{r} \amp = \int_{-1}^{0} (e^0 \sin(-t), e^0 \cos(-t) + 0) \cdot (0, -1) \, dt\\ \amp = \int_{-1}^{0} -\cos(-t) \, dt = -\int_{-1}^{0} \cos t \, dt = -\sin t \Big|_{-1}^{0}\\ \amp = \sin(-1) = -\sin(1) \end{align*}
